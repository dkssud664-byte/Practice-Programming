


# C#으로 List 구현방법

구조 설명 : 템플릿 배열을 이용한 구현

만드는 순서 : 클래스 만들기 - 변수 선언 - 프로퍼티 구현 - 생성자 구현 - 인덱서 구현 - Add() - Grow() - RemoveAt() - Remove() - Insert() - IndexOf() - Clear()

1. List.cs 만들기, 템플릿 클래스로 변환
```csharp
public class List<T>
```
<br>

2. 실제로 담을 변수 선언
```csharp
private const int DefaultCapacity = 4;      //리스트가 처음 만들어질 때 기본 크기

private T[] items;      //실제 데이터를 저장하는 배열
private int count;      //리스트 안에 현재 들어있는 데이터 개수
```

<br>

3. private 변수를 꺼내 쓰기위한 프로퍼티 선언
```csharp
public int Capacity { get { return items.Length; } }
public int Count {  get { return count; } }
```

4. 메모리 할당할 수 있게 생성자 만들기
```csharp
public List()
```

5. 데이터를 저장할 T[]의 특정 인덱스에 접근할수 있게 인덱서 선언

	- 인덱서는 클래스나 구조체를 배열처럼 사용할 수 있게 해주는 기능이다.
```csharp
//인덱서(indexer) : 리스트를 배열처럼 사용할 수 있게 해줌
public T this[int index]
{
    get
    {
        return items[index];
    }
    set
    {
        items[index] = value;
    }
}
```

6. 데이터를 넣기 위해 Add 구현
```csharp
public void Add(T item)
{
    if(count == items.Length)
    {
        Grow();
    }

    items[count] = item;
    count++;
}
```
<br>

7. 배열이 초과했을 때를 위해 Grow() 구현
```csharp
//내부 배열 크기를 2배로 늘림
private void Grow()
{
    T[] newItems = new T[items.Length * 2];     //2배!
    Array.Copy(items, 0, newItems, 0, count);
    items = newItems;
}
```
<br>

8. 특정 배열을 지우기 위해 RemoveAt() 구현
```csharp
 public void RemoveAt(int index)
 {
     if (index < 0 || count <= index)
         throw new IndexOutOfRangeException();

     count--;
     Array.Copy(items, index + 1, items, index, count - index);
 }
```

9. 특정 값을 지우기 위해 Remove() 구현
```csharp
public bool Remove(T item)
{
    int index = IndexOf(item);

    if (index >= 0)
    {
        RemoveAt(index);
        return true;
    }
    else
    {
        return false;
    }
}
```

10. 인덱스 중간에 값을 넣기 위해 Insert() 구현
```csharp
 //원하는 위치(index)에 새로운 자료(또는 요소) 삽입
 public void Insert(int index, T item)
 {
     if (index < 0 || count < index)
         throw new ArgumentOutOfRangeException();    //예외처리 발생시 알려줌

     // 인덱스가 카운트와 같다는 말은! 맨 뒤에 자료를 삽입하는 경우인데 이땐 Add와 다를바가 없다
     if(index == count)
     {
         Add(item);
         return;
     }

     //배열이 꽉 찼다면 크기를 늘려줘라
     if(count == items.Length)
     {
         Grow();
     }

     //삽입 위치 뒤의 자료들을 한 칸씩 뒤로 밀어준다 (왜 지금 중간에 자료 삽입 메서드를 구현중이니까)
     Array.Copy(items, index, items, index + 1, count - index);

     //원하는 위치에 데이터를 넣기
     items[index] = item;
     count++;
 }
```

<br>

11. 값의 인덱스 위치를 알기 위해 IndexOf() 구현
```csharp
public int IndexOf(T item)
{
    if(item != null)
    {
        for(int i = 0; i < count; i++)
        {
            if(item.Equals(items[i]))
            {
                return i;       //찾았으면 위치 반환
            }
        }
    }
    else
    {
        for(int i = 0; i < count; i++)
        {
            if(null == items[i])
            {
                return i;
            }
        }
    }

    return -1;
}
```

12. List를 초기화 하기 위해 Clear() 구현
```csharp
 //리스트 완전히 비우는 메서드
 public void Clear()
 {
     items = new T[DefaultCapacity]; //새 배열 생성
     count = 0;                      //개수도 초기화
 }
```

**총 코드**
```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace _01.List
{
    public class List<T>
    {
        private const int DefaultCapacity = 4;      //리스트가 처음 만들어질 때 기본 크기

        private T[] items;      //실제 데이터를 저장하는 배열
        private int count;      //리스트 안에 현재 들어있는 데이터 개수

        public List()
        {
            items = new T[DefaultCapacity];
            count = 0;
        }

        public int Capacity { get { return items.Length; } }
        public int Count {  get { return count; } }

        //인덱서(indexer) : 리스트를 배열처럼 사용할 수 있게 해줌
        public T this[int index]
        {
            get
            {
                return items[index];
            }
            set
            {
                items[index] = value;
            }
        }

        //자료(또는 요소) 추가 메서드
        public void Add(T item)
        {
            if(count == items.Length)
            {
                Grow();
            }

            items[count] = item;
            count++;
        }

        //원하는 위치(index)에 새로운 자료(또는 요소) 삽입
        public void Insert(int index, T item)
        {
            if (index < 0 || count < index)
                throw new ArgumentOutOfRangeException();    //예외처리 발생시 알려줌

            // 인덱스가 카운트와 같다는 말은! 맨 뒤에 자료를 삽입하는 경우인데 이땐 Add와 다를바가 없다
            if(index == count)
            {
                Add(item);
                return;
            }

            //배열이 꽉 찼다면 크기를 늘려줘라
            if(count == items.Length)
            {
                Grow();
            }

            //삽입 위치 뒤의 자료들을 한 칸씩 뒤로 밀어준다 (왜 지금 중간에 자료 삽입 메서드를 구현중이니까)
            Array.Copy(items, index, items, index + 1, count - index);

            //원하는 위치에 데이터를 넣기
            items[index] = item;
            count++;
        }

        public bool Remove(T item)
        {
            int index = IndexOf(item);

            if (index >= 0)
            {
                RemoveAt(index);
                return true;
            }
            else
            {
                return false;
            }
        }

        public void RemoveAt(int index)
        {
            if (index < 0 || count <= index)
                throw new IndexOutOfRangeException();

            count--;
            Array.Copy(items, index + 1, items, index, count - index);

        }

        //특정 값이 몇 번째 인덱스에 있는지 찾아서 반환해주는 메서드
        public int IndexOf(T item)
        {
            if(item != null)
            {
                for(int i = 0; i < count; i++)
                {
                    if(item.Equals(items[i]))
                    {
                        return i;       //찾았으면 위치 반환
                    }
                }
            }
            else
            {
                for(int i = 0; i < count; i++)
                {
                    if(null == items[i])
                    {
                        return i;
                    }
                }
            }

            return -1;
        }

        //리스트 완전히 비우는 메서드
        public void Clear()
        {
            items = new T[DefaultCapacity]; //새 배열 생성
            count = 0;                      //개수도 초기화
        }

        //내부 배열 크기를 2배로 늘림
        private void Grow()
        {
            T[] newItems = new T[items.Length * 2];     //2배!
            Array.Copy(items, 0, newItems, 0, count);
            items = newItems;
        }
    }
}
```
