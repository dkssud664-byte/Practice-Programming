using System;

using System.Collections.Generic;

using System.Linq;

using System.Text;

using System.Threading.Tasks;



namespace \_01.List

{

&nbsp;   public class List<T>

&nbsp;   {

&nbsp;       private const int DefaultCapacity = 4;      //리스트가 처음 만들어질 때 기본 크기



&nbsp;       private T\[] items;      //실제 데이터를 저장하는 배열

&nbsp;       private int count;      //리스트 안에 현재 들어있는 데이터 개수



&nbsp;       public List()

&nbsp;       {

&nbsp;           items = new T\[DefaultCapacity];

&nbsp;           count = 0;

&nbsp;       }



&nbsp;       public int Capacity { get { return items.Length; } }

&nbsp;       public int Count {  get { return count; } }



&nbsp;       //인덱서(indexer) : 리스트를 배열처럼 사용할 수 있게 해줌

&nbsp;       public T this\[int index]

&nbsp;       {

&nbsp;           get

&nbsp;           {

&nbsp;               return items\[index];

&nbsp;           }

&nbsp;           set

&nbsp;           {

&nbsp;               items\[index] = value;

&nbsp;           }

&nbsp;       }



&nbsp;       //자료(또는 요소) 추가 메서드

&nbsp;       public void Add(T item)

&nbsp;       {

&nbsp;           if(count == items.Length)

&nbsp;           {

&nbsp;               Grow();

&nbsp;           }



&nbsp;           items\[count] = item;

&nbsp;           count++;

&nbsp;       }



&nbsp;       //원하는 위치(index)에 새로운 자료(또는 요소) 삽입

&nbsp;       public void Insert(int index, T item)

&nbsp;       {

&nbsp;           if (index < 0 || count < index)

&nbsp;               throw new ArgumentOutOfRangeException();    //예외처리 발생시 알려줌



&nbsp;           // 인덱스가 카운트와 같다는 말은! 맨 뒤에 자료를 삽입하는 경우인데 이땐 Add와 다를바가 없다

&nbsp;           if(index == count)

&nbsp;           {

&nbsp;               Add(item);

&nbsp;               return;

&nbsp;           }



&nbsp;           //배열이 꽉 찼다면 크기를 늘려줘라

&nbsp;           if(count == items.Length)

&nbsp;           {

&nbsp;               Grow();

&nbsp;           }



&nbsp;           //삽입 위치 뒤의 자료들을 한 칸씩 뒤로 밀어준다 (왜 지금 중간에 자료 삽입 메서드를 구현중이니까)

&nbsp;           Array.Copy(items, index, items, index + 1, count - index);



&nbsp;           //원하는 위치에 데이터를 넣기

&nbsp;           items\[index] = item;

&nbsp;           count++;

&nbsp;       }



&nbsp;       public bool Remove(T item)

&nbsp;       {

&nbsp;           int index = IndexOf(item);



&nbsp;           if (index >= 0)

&nbsp;           {

&nbsp;               RemoveAt(index);

&nbsp;               return true;

&nbsp;           }

&nbsp;           else

&nbsp;           {

&nbsp;               return false;

&nbsp;           }

&nbsp;       }



&nbsp;       public void RemoveAt(int index)

&nbsp;       {

&nbsp;           if (index < 0 || count <= index)

&nbsp;               throw new IndexOutOfRangeException();



&nbsp;           count--;

&nbsp;           Array.Copy(items, index + 1, items, index, count - index);



&nbsp;       }



&nbsp;       //특정 값이 몇 번째 인덱스에 있는지 찾아서 반환해주는 메서드

&nbsp;       public int IndexOf(T item)

&nbsp;       {

&nbsp;           if(item != null)

&nbsp;           {

&nbsp;               for(int i = 0; i < count; i++)

&nbsp;               {

&nbsp;                   if(item.Equals(items\[i]))

&nbsp;                   {

&nbsp;                       return i;       //찾았으면 위치 반환

&nbsp;                   }

&nbsp;               }

&nbsp;           }

&nbsp;           else

&nbsp;           {

&nbsp;               for(int i = 0; i < count; i++)

&nbsp;               {

&nbsp;                   if(null == items\[i])

&nbsp;                   {

&nbsp;                       return i;

&nbsp;                   }

&nbsp;               }

&nbsp;           }



&nbsp;           return -1;

&nbsp;       }



&nbsp;       //리스트 완전히 비우는 메서드

&nbsp;       public void Clear()

&nbsp;       {

&nbsp;           items = new T\[DefaultCapacity]; //새 배열 생성

&nbsp;           count = 0;                      //개수도 초기화

&nbsp;       }



&nbsp;       //내부 배열 크기를 2배로 늘림

&nbsp;       private void Grow()

&nbsp;       {

&nbsp;           T\[] newItems = new T\[items.Length \* 2];     //2배!

&nbsp;           Array.Copy(items, 0, newItems, 0, count);

&nbsp;           items = newItems;

&nbsp;       }

&nbsp;   }

}
