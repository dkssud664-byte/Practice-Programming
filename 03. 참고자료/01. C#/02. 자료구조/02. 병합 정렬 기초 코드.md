using System;
using System.Collections.Generic;

public class MergeSortBasic
{
    public static void MergeSort(List<int> list)
    {
        // 원소 하나면 이미 정렬됨
        if (list.Count <= 1)
            return;

        int mid = list.Count / 2;

        // 왼쪽, 오른쪽 리스트 분리
        List<int> left = new List<int>();
        List<int> right = new List<int>();

        for (int i = 0; i < mid; i++)
            left.Add(list[i]);

        for (int i = mid; i < list.Count; i++)
            right.Add(list[i]);

        // 재귀적으로 반씩 정렬
        MergeSort(left);
        MergeSort(right);

        // 두 리스트를 합쳐서 정렬
        List<int> merged = Merge(left, right);

        // 결과를 원래 list에 복사
        for (int i = 0; i < list.Count; i++)
            list[i] = merged[i];
    }

    private static List<int> Merge(List<int> left, List<int> right)
    {
        List<int> result = new List<int>();
        int li = 0, ri = 0;

        // 둘 다 비어있지 않을 동안 작은 값 넣기
        while (li < left.Count && ri < right.Count)
        {
            if (left[li] <= right[ri])
                result.Add(left[li++]);
            else
                result.Add(right[ri++]);
        }

        // 남은 부분 전부 추가
        while (li < left.Count)
            result.Add(left[li++]);

        while (ri < right.Count)
            result.Add(right[ri++]);

        return result;
    }
}
