
# C# 순차탐색 코드

```csharp
// <순차 탐색>
// 자료구조에서 순차적으로 찾고자 하는 데이터를 탐색
// 시간복잡도 - O(n)

public static int SequentialSearch<T>(IList<T> list, in T item) where T : notnull
{
    for(int i = 0; i < list.Count; i++)
    {
        if (list[i].Equals(item))
        {
            return i;
        }
    }
    return -1;
}
```
