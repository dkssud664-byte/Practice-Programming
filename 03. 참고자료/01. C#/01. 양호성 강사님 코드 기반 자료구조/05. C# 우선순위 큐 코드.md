
# C# 우선순위 큐 코드

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace DataStructure
{
    // 제네릭 우선순위 큐 (Priority Queue) 구현
    // 요소(TElement)와 우선순위(Tpriority)를 함께 저장
    // 우선순위가 낮을 수록 높은 우선순위를 가짐 ( 최소 힙 Min Heap 구조)
    // 내부적으로 이진 힙(Binary Heap)을 List<Node>로 구현 

    public class PriorityQueue<TElement, TPriority> where TPriority : IComparable<TPriority>
    {
        private struct Node
        {
            public TElement element;
            public TPriority priority;

            public Node(TElement element, TPriority priority)
            {
                this.element = element;
                this.priority = priority;
            }
        }

        //힙 구조를 저장할 동적배열(List)
        // 인덱스 0이 루트 노드가 된다 
        private List<Node> nodes;

        public PriorityQueue()
        {
            nodes = new List<Node>();
        }

        //프로퍼티
        public int Count { get { return nodes.Count; } }

        //새로운 요소를 우선순위 함께 큐에 추가
        //요소를 리스트 끝에 추가한 후, 상향식으로 힙 속성 복원 (Heapify Up)
        public void Enqueue(TElement element, TPriority priority)
        {
            Node node = new Node(element, priority);
            nodes.Add(node);

            int index = nodes.Count - 1;

            //부모 노드와 비교해서 올라가며 힙 속성 복원 (상향식 정렬)
            while(index > 0)
            {
                int parentIndex = GetParentIndex(index);
                Node parentNode = nodes[parentIndex];

                //현재 노드의 우선순위가 부모보다 높거나 같으면 정지
                //CompareTo: 작을수록 -1    -> 우선순위 높음 
                if (node.priority.CompareTo(parentNode.priority) >= 0) break;

                //부모가 더 낮은 우선순위(더 큰 값) 이면, 부모와 위치 교환
                nodes[index] = parentNode;
                nodes[parentIndex] = node;
                index = parentIndex;    //현재 노드를 부모 위치로 이동
            }
        }

        public TElement Peek()
        {
            if (nodes.Count == 0)
                throw new InvalidOperationException();

            return nodes[0].element;
        }
        
        //최고 우선순위 요소를 제거하고 반환
        // - 루트 노드를 제거하고, 마지막 노드를 루트로 이동
        // - 하향식으로 힙 속성 복원 (Heapify Down)
        public TElement Dequeue()
        {
            if (nodes.Count == 0)
                throw new InvalidOperationException();

            Node rootNode = nodes[0];           //제거할 루트 노드 (최고 우선순위)
            Node node = nodes[nodes.Count - 1]; //리스트의 마지막 노드 
            nodes[0] = node;                    //마지막 노드를 루트 위치로 이동
            nodes.RemoveAt(nodes.Count - 1);    //마지막 노드 제거 

            int index = 0;

            //하향식 힙 속성 복원 
            while (index < nodes.Count)
            {
                int leftIndex = GetLeftChildIndex(index);
                int rightIndex = GetRightChildIndex(index);
                int targetIndex = index;

                if(leftIndex < Count &&
                    nodes[leftIndex].priority.CompareTo(nodes[targetIndex].priority) < 0)
                {
                    targetIndex = leftIndex;
                }

                if(rightIndex < Count &&
                    nodes[rightIndex].priority.CompareTo(nodes[targetIndex].priority) < 0)
                {
                    targetIndex = rightIndex;
                }

                //더 이상 교환할 자식이 없으면 (힙 속성이 만족됐다면) -> 종료
                if (index == targetIndex) break;

                //현재 노드와 우선순위가 더 높은 자식 교환
                nodes[index] = nodes[targetIndex];
                nodes[targetIndex] = node;
                index = targetIndex;            //교환된 위치로 이동해서 계속 정렬
            }

            return rootNode.element;
        }


        private int GetParentIndex(int index)
        {
            return (index - 1) / 2;
        }

        private int GetLeftChildIndex(int index)
        {
            return index * 2 + 1;
        }

        private int GetRightChildIndex(int index)
        {
            return index * 2 + 2;
        }
    }
}

```
