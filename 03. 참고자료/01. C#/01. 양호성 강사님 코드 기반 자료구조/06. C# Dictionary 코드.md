
# C# Dictionary 코드

```csharp
using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace DataStructure
{
    // where TKey : IEquatable<TKey> -> 키는 서로 비교할 수 있어야 함 (키 값 중복여부)
    public class Dictionary<TKey, TValue> where TKey : IEquatable<TKey>
    {
        //기본 테이블 크기 : 1000개 먼저 만들어둠
        private const int DefaultCapacity = 1000;

        //하나의 칸을 나타내는 구조체
        // 각 칸은 "비어있음", "사용중", "삭제됨" 상태를 가질 수 있다 
        private struct Entry
        {
            public enum State { None, Using, Deleted }

            public State state;
            public TKey key;
            public TValue value;
        }

        private Entry[] table;          //실제 데이터를 저장하는 배열
        private int usedCount;          //사용중인 칸 개수
        private int deletedCount;       //삭제된 칸 개수 

        public Dictionary()
        {
            table = new Entry[DefaultCapacity];
            usedCount = 0;
            deletedCount = 0;
        }

        public TValue this[TKey key]
        {
            get
            {
                if (TryFind(key, out int index) == false)
                    throw new KeyNotFoundException();

                return table[index].value;
            }
            set
            {
                if(TryFind(key, out int index))
                {
                    table[index].value = value;
                }
                else
                {
                    Add(key, value);
                }
            }
        }

        //테이블 크기
        public int Capacity { get { return table.Length; } }
        public int Count { get { return usedCount; } }

        // 새 데이터 추가 (키 중복되면 에러남!)
        public void Add(TKey key, TValue value)
        {
            // 이미 있는 키인지 확인 -> 있으면 에러
            if (TryFind(key, out int index) == false)
                throw new InvalidOperationException();

            //테이블 70% 이상 차면 (테이블이 꽉 차려고 하면) -> 방을 2배로 확장한다
            if(usedCount + deletedCount > table.Length * 0.7f)
            {
                ReHashing();
            }

            //위에 예외사항이 아니면 빈 칸을 찾았을테니 데이터 넣기 
            table[index].key = key;
            table[index].value = value;
            table[index].state = Entry.State.Using;
            usedCount++;
        }

        //모든 데이터 지우기 -> 새 테이블로 교체
        public void Clear()
        {
            table = new Entry[DefaultCapacity];
            usedCount = 0;
            deletedCount = 0;
        }

        public bool ContainsKey(TKey key)
        {
            if(TryFind(key, out int index))
            {
                return true;
            }
            else
            {
                return false;
            }
        }

        //키로 데이터 삭제
        public bool Remove(TKey key)
        {
            if(TryFind(key, out int index))
            {
                table[index].state = Entry.State.Deleted;   //바로 지우지 않고 "삭제표시" 만 해둔다 
                usedCount--;
                deletedCount++;
                return true;
            }
            else
            {
                return false;
            }
        }

        //거의 핵심 메서드가 아닐까 (키를 주면 그 키가 있는 위치 (index)를 찾아줌)
        //out int index -> 찾은 위치를 반환하기 위해 사용한 매개변수
        private bool TryFind(TKey key, out int index)
        {
            if (key == null)
                throw new ArgumentNullException();

            //1. 키를 해시 함수로 숫자 (방 번호)로 바꿈
            index = Hash(key);

            //2. 테이블 끝날 때까지 탐색 (충돌 해결 == 선형 탐사)
            for(int i = 0; i < table.Length; i++)
            {
                //빈 칸을 만났다? -> 없다는 뜻
                if (table[index].state == Entry.State.None)
                {
                    return false;
                }
                else if (table[index].state == Entry.State.Using)
                {
                    if(key.Equals(table[index].key))
                    {
                        return true;
                    }
                    else
                    {
                        index = Hash2(index);
                    }
                }
                else // if (table[index].state == Entry.State.Deleted)
                {
                    index = Hash2(index);
                }
            }

            index = -1;
            throw new InvalidOperationException();
        }

        // 해시 함수 : 키를 숫자로 변환
        private int Hash(TKey key)
        {
            return Math.Abs(key.GetHashCode() % table.Length);
        }

        private int Hash2(int index)
        {
            //선형 탐사 (한 칸씩 (가장 단순))
            return (index + 1) % table.Length;

            //제곱 탐사 (점점 더 멀리)
            //return (index + 1) * (index + 1) % table.Length;

            //이중 해싱 (키 자체를 다시 해싱)
            //return Math.Abs((index + 1).GetHashCode() % table.Length);
        }

        //테이블 확장 + 재배치 메서드 
        private void ReHashing()
        {
            Entry[] oldTable = table;
            table = new Entry[table.Length * 2];
            usedCount = 0;
            deletedCount = 0;

            for(int i = 0; i < oldTable.Length; i++)
            {
                Entry entry = oldTable[i];
                if(entry.state == Entry.State.Using)
                {
                    Add(entry.key, entry.value);
                    //Add 메서드 만들고 넣어줍시다 
                }
            }
        }
    }
}
```
