using System;

public class Solution {
    public int[] solution(int[,] arr)
    {
        int[] answer = new int[2];
        int maxSum = arr.GetLength(0) * arr.GetLength(1);
        int sum = 0;

        for(int i = 0; i < arr.GetLength(0); i++)
        {
            for(int j = 0; j < arr.GetLength(1); j++)
            {
                sum += arr[i, j];
            }
        }

        //모든 값이 동일할경우
        if(sum == 0 || sum == maxSum)
        {
            if(sum == 0)
            {
                answer[0]++;
            }
            else
            {
                answer[1]++;
            }
            return answer;
        }
        
        

        RectangleArray(arr, ref answer);

        return answer;
    }

    public void RectangleArray(int[,] arr, ref int[] answer)
    {

        //탈출 조건
        if (arr.GetLength(0) < 2)
        {
            if (arr[0, 0] == 0)
            {
                answer[0]++;
            }
            else
            {
                answer[1]++;
            }
        }

        //사각형 4개
        int[,,] half = new int[4, arr.GetLength(0) / 2, arr.GetLength(1) / 2];
        int halfZ = 0;
        int halfY = 0;
        int halfX = 0;

        //반으로 자른 배열 값 넣기
        for (int i = 0; i < arr.GetLength(0); i++)
        {
            for(int j = 0; j < arr.GetLength(1) / 2; j++)
            {
                //왼쪽
                half[halfZ, halfY, halfX++] = arr[i, j];

                if(halfX >= half.GetLength(2))
                {
                    halfX = 0;
                    halfY++;
                    if (halfY >= half.GetLength(1))
                    {
                        halfY = 0;
                        halfZ++;
                    }
                }
            }

        }

        halfY = 0;
        halfX = 0;

        //반으로 자른 배열 값 넣기
        for (int i = 0; i < arr.GetLength(0); i++)
        {
            for (int k = arr.GetLength(1) / 2; k < arr.GetLength(1); k++)
            {
                //오른쪽
                half[halfZ, halfY, halfX++] = arr[i, k];

                if (halfX >= half.GetLength(2))
                {
                    halfX = 0;
                    halfY++;
                    if (halfY >= half.GetLength(1))
                    {
                        halfY = 0;
                        halfZ++;
                    }
                }
            }
        }

        //숫자 세기
        for (int i = 0; i < half.GetLength(0); i++)
        {
            int[,] temp = new int[half.GetLength(1), half.GetLength(2)];

            for(int j = 0; j < half.GetLength(1); j++)
            {
                for(int k = 0; k < half.GetLength(2); k++)
                {
                    temp[j, k] = half[i, j, k];
                }
            }

            if(IsSameNumberArray(temp))
            {
                if (temp[0,0] == 0)
                {
                    answer[0]++;
                }
                else
                {
                    answer[1]++;
                }
                continue;
            }

            RectangleArray(temp, ref answer);
        }

    }

    //사각형 안의 같은 값 확인
    public bool IsSameNumberArray(int[,] arr)
    {
        int maxSum = arr.GetLength(0) * arr.GetLength(1);
        int sum = 0;
        int temp = arr[0, 0];

        for(int i = 0; i < arr.GetLength(0); i++)
        {
            for(int j = 0; j < arr.GetLength(1); j++)
            {
                sum += arr[i, j];
                if (temp != arr[i, j])
                {
                    return false;
                }
                temp = arr[i, j];
            }
        }

        return true;
    }
}
